$$\mathcal{R} = \{\text{SWI/SNF}, \text{ISWI}, \text{CHD}, \text{INO80}\}$$

$$\vec{v}_{\text{slide}} = \frac{k_{\text{ATP}} \cdot [\text{ATP}]}{k_{\text{ATP}} \cdot [\text{ATP}] + K_m} \cdot \vec{F}_{\text{DNA}}$$
