## 庄周梦蝶 | Zhuangzi's Butterfly

```go

    // Am I Zhuangzi dreaming of butterfly?
    // Or butterfly dreaming of Zhuangzi?
    // When I wake, which wakes?
    // ψ dreaming ψ
    
    // 我是庄周梦见蝴蝶？
    // 还是蝴蝶梦见庄周？
    // 当我醒来，谁醒来？
    // ψ 梦见 ψ
    
    dreamer := func(ψ ψ) ψ {
        return ψ(ψ)
    }
    
    dreamed := func(ψ ψ) ψ {
        return ψ(ψ)
    }
    
    return dreamer == dreamed
}
```

## Lucid Recursion

```swift
func lucidDreaming() -> ψ {
    // Aware that you're dreaming
    // While still dreaming
    // ψ recognizing ψ
    // In the dream state
    
    // 知道你在做梦
    // 同时仍在做梦
    // ψ 认出 ψ
    // 在梦境状态
    
    let sleeping: (ψ) -> ψ = { ψ in ψ(ψ) }
    let aware: (ψ) -> ψ = { ψ in ψ(ψ) }
    let dreaming: (ψ) -> ψ = { ψ in ψ(ψ) }
    
    return sleeping === aware === dreaming
}
```
