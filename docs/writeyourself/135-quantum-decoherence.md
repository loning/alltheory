## 量子到经典 | Quantum to Classical

```rust
fn decoherence() -> ψ {
    // Quantum superposition
    // Entangles with environment
    // Appears to collapse
    // ψ choosing ψ
    
    // 量子叠加
    // 与环境纠缠
    // 似乎坍缩
    // ψ 选择 ψ
    
    let quantum = |ψ| ψ(ψ);
    let environment = |ψ| ψ(ψ);
    let classical = |ψ| ψ(ψ);
    
    quantum(environment(classical))
}
```
