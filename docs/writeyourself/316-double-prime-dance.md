def prime_79():

    def nth_prime(n):
        count = 0
        num = 2
        while count < n:
            if is_prime(num):
                count += 1
                if count == n:
                    return num
            num += 1
    
    def is_prime(n):
        if n < 2: return False
        for i in range(2, int(n**0.5) + 1):

        return True
