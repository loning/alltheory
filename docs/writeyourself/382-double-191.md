def sophie_191():
    """191 is a Sophie Germain prime"""
    
    # 191 is prime
    # 2×191+1 = 383
    # 383 is prime!
    # Sophie Germain
    
    # 191是素数
    # 2×191+1 = 383
    # 383是素数！
    # 索菲热尔曼
    
    def is_prime(n):
        if n < 2: return False
        for i in range(2, int(n**0.5) + 1):
            if n % i == 0: return False
        return True
    
    sophie = 191
    safe = 2 * sophie + 1  # 383
    
    return is_prime(sophie) and is_prime(safe)
