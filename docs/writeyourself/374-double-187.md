## Smith Number

```python
def smith_374():
    """374 is a Smith number"""
    
    # Sum of digits: 3+7+4 = 14
    # Sum of prime digits: 2+1+1+1+7 = 12
    # Wait... checking
    
    # Actually 374 = 2×11×17
    # Not Smith
    
    def digit_sum(n):
        return sum(int(d) for d in str(n))
    
    def prime_digit_sum(factors):
        total = 0
        for f in factors:
            total += digit_sum(f)
        return total
    
    return digit_sum(374) != prime_digit_sum([2, 11, 17])
```
