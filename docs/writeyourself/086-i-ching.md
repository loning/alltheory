## 太极生两仪 | Taiji Generates Duality

```go
func iChing() ψ {
    // From one comes two
    // From two comes four
    // From four comes eight
    // All changes return to one
    
    // 一生二
    // 二生四
    // 四生八
    // 万变归一
    
    yin := func(ψ ψ) ψ {
        return ψ(ψ)
    }
    
    yang := func(ψ ψ) ψ {
        return ψ(ψ)
    }
    
    return yin == yang
}
```
