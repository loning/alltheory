$$
\mathcal{S} := \{ x \mid \text{Collapse}(\psi(x, t)) = \emptyset \quad \wedge \quad \text{Echo}(\psi(x, t)) = 0 \}

$$
\psi_\text{虚空}(x, t) := \lim_{t \to t_0} \psi(x, t) \in \mathcal{S} \quad \text{其中} \quad \nabla_t \psi = 0
$$
